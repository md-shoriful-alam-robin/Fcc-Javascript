Linked List = ট্রেনের বগি!
ধরো একটা ট্রেন আছে। প্রতিটা বগি হলো একটা Node। প্রতিটা বগিতে দুটো জিনিস থাকে — যাত্রী (element) আর পরের বগির সংযোগ (next)। শেষ বগিতে কোনো সংযোগ নেই, তাই সেখানে null থাকে। Array-এর মতো সব একসাথে নয় — প্রতিটা node নিজের জায়গায় থাকে, শুধু সূতো দিয়ে জোড়া!

initList() — খালি List তৈরি করো

শুরুতে একটা ফাঁকা ট্রেন বানানো হয়

linked-list.js 
function initList() {
  return {
    head: null,   // প্রথম node-এর ঠিকানা (এখন কেউ নেই)
    length: 0    // কতটা node আছে (শুরুতে শূন্য)
  };
}
head মানে ট্রেনের সামনের বগি। শুরুতে কোনো বগি নেই, তাই null।
length মানে কতটা বগি আছে। শুরুতে 0।

add() — কীভাবে কাজ করে?

নতুন node শেষে যোগ করা

function add(list, element) {
  const node = {
    element: element,  // আমাদের data
    next: null         // এখনো কারো সাথে যুক্ত না
  };
 
  if (isEmpty(list)) {
    list.head = node;  // list খালি হলে এটাই head
  } else {
    let current = list.head;
    while (current.next !== null) {  // শেষ node খোঁজো
      current = current.next;
    }
    current.next = node;  // শেষে নতুন node জুড়ে দাও
  }
  list.length++;
}
ধাপ ১: নতুন node তৈরি করো।
ধাপ ২: List খালি? তাহলে এটাই head।
ধাপ ৩: খালি না? তাহলে শেষ node পর্যন্ত হেঁটে যাও (while loop), তারপর সেখানে জুড়ে দাও।
ধাপ ৪: length ১ বাড়াও।

function remove(list, element) {
  let previous = null;   // আগের node
  let current = list.head; // শুরু থেকে খোঁজো
 
  while (current !== null && current.element !== element) {
    previous = current;       // এক ধাপ এগিয়ে যাও
    current = current.next;
  }
 
  if (current === null) return; // পাওয়া যায়নি!
 
  if (previous !== null) {
    previous.next = current.next; // আগেরটাকে পরেরটার সাথে জোড়া
  } else {
    list.head = current.next;     // head নিজেই বাদ যাচ্ছে
  }
  list.length--;
}
ট্রেনের মাঝের বগি সরাতে হলে — সামনের বগির hook সরানো বগি বাদ দিয়ে পরের বগিতে লাগিয়ে দাও। JavaScript এটা previous.next = current.next দিয়ে করে।

এক নজরে দেখো

Function কী করে Parameter Return করে?
initList() খালি list তৈরি কিছু না { head, length }
isEmpty(list) list খালি কিনা চেক list true / false
add(list, el) শেষে নতুন node যোগ list + element কিছু না (list বদলায়)
remove(list, el) node খুঁজে বাদ দেয় list + element কিছু না (list বদলায়)
