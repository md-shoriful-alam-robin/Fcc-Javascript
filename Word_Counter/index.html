getMatchedWordCount.js Array → count matches
function getMatchedWordCount(sentence, match) {
  let count = 0;
  for (const word of sentence) {
    if (word === match) {
      count++;
    }
    console.log(`Checking "${word}" vs "${match}" | count: ${count}`);
  }
  return count;
}
